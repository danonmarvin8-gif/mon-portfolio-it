Schéma de la Base de Données (MySQL)

-- Tables relationnelles de l'application CREATE TABLE vehicules ( id_vehicule INT PRIMARY KEY AUTO_INCREMENT, immatriculation VARCHAR(15) UNIQUE NOT NULL, modele VARCHAR(50) NOT NULL, capacite INT NOT NULL, statut ENUM('disponible', 'en_trajet', 'maintenance') DEFAULT 'disponible' ); CREATE TABLE chauffeurs ( id_chauffeur INT PRIMARY KEY AUTO_INCREMENT, nom VARCHAR(50) NOT NULL, prenom VARCHAR(50) NOT NULL, telephone VARCHAR(15), statut ENUM('disponible', 'conge', 'en_mission') DEFAULT 'disponible' ); CREATE TABLE reservations ( id_reservation INT PRIMARY KEY AUTO_INCREMENT, nom_client VARCHAR(100) NOT NULL, destination VARCHAR(100) NOT NULL, date_depart DATETIME NOT NULL, id_vehicule INT, id_chauffeur INT, statut ENUM('en_attente', 'confirmee', 'annulee') DEFAULT 'confirmee', FOREIGN KEY (id_vehicule) REFERENCES vehicules(id_vehicule), FOREIGN KEY (id_chauffeur) REFERENCES chauffeurs(id_chauffeur) );
