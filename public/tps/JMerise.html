Extrait SQL généré

CREATE TABLE GRADE ( id_grade INT PRIMARY KEY AUTO_INCREMENT, libelle_grade VARCHAR(50) NOT NULL, niveau_hierarchique INT, code_grade VARCHAR(10) UNIQUE ); CREATE TABLE MILITAIRE ( matricule VARCHAR(10) PRIMARY KEY, nom VARCHAR(80) NOT NULL, prenom VARCHAR(80) NOT NULL, date_naissance DATE, date_engagement DATE NOT NULL, id_grade INT, id_unite INT, FOREIGN KEY (id_grade) REFERENCES GRADE(id_grade), FOREIGN KEY (id_unite) REFERENCES UNITE(id_unite) ); CREATE TABLE PARTICIPE ( matricule VARCHAR(10), id_mission INT, role_mission VARCHAR(100), PRIMARY KEY (matricule, id_mission), FOREIGN KEY (matricule) REFERENCES MILITAIRE(matricule), FOREIGN KEY (id_mission) REFERENCES MISSION(id_mission) );
